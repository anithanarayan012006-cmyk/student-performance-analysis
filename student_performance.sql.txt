-- ============================================
-- DATA ANALYSIS QUERIES
-- ============================================

-- 1. View all students
SELECT * FROM students;

-- 2. Average marks
SELECT AVG(marks) AS average_marks
FROM students;

-- 3. Highest-scoring student
SELECT student_name, subject, marks
FROM students
ORDER BY marks DESC
LIMIT 1;

-- 4. Lowest-scoring student
SELECT student_name, subject, marks
FROM students
ORDER BY marks ASC
LIMIT 1;

-- 5. Average marks by subject
SELECT subject, AVG(marks) AS average_marks
FROM students
GROUP BY subject
ORDER BY average_marks DESC;

-- 6. Lowest attendance
SELECT student_name, attendance
FROM students
ORDER BY attendance ASC
LIMIT 1;

-- 7. Students with attendance below 80%
SELECT student_name, marks, attendance
FROM students
WHERE attendance < 80
ORDER BY attendance ASC;

-- 8. Students scoring above 80
SELECT student_name, subject, marks
FROM students
WHERE marks > 80
ORDER BY marks DESC;

-- 9. Percentage scoring above 80
SELECT
    COUNT(*) * 100.0 / (SELECT COUNT(*) FROM students)
    AS percentage_above_80
FROM students
WHERE marks > 80;

-- 10. Average attendance
SELECT AVG(attendance) AS average_attendance
FROM students;

-- 11. Top 3 students
SELECT student_name, subject, marks, attendance
FROM students
ORDER BY marks DESC
LIMIT 3;

-- 12. Students ordered by attendance
SELECT student_name, marks, attendance
FROM students
ORDER BY attendance DESC;
